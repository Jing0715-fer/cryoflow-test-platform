   147→export function readRuns(): Record<string, RunRecord> {
   148→  // mtime-keyed cache: EVERY polled route (jobs GET, log, outputs, particles,
   149→  // …) used to re-read + re-JSON.parse the whole state file on every call —
   150→  // the file grows with every job ever run and the parse ran on the hot 2–5 s
   151→  // poll path. writeRuns bumps the mtime so writers are always consistent.
   152→  try {
   153→    const st = statSync(STATE_FILE);
   154→    if (runsCache && runsCache.mtime === st.mtimeMs && runsCache.size === st.size) {
   155→      return runsCache.value;
   156→    }
   157→    const parsed = JSON.parse(readFileSync(STATE_FILE, "utf8"));
   158→    if (parsed && typeof parsed === "object" && !Array.isArray(parsed)) {
   159→      const value = parsed as Record<string, RunRecord>;
   160→      runsCache = { mtime: st.mtimeMs, size: st.size, value };
   161→      return value;
   162→    }
   163→  } catch {
   164→    // ENOENT / corrupt → fresh state
   165→    runsCache = null;
   166→  }
   167→  return {};
   168→}
   169→
   170→let runsCache: { mtime: number; size: number; value: Record<string, RunRecord> } | null = null;

   172→export function writeRuns(map: Record<string, RunRecord>): void {
   173→  mkdirSync(DATA_DIR, { recursive: true });
   174→  writeFileSync(STATE_FILE, JSON.stringify(map, null, 2));
   175→  // invalidate immediately — the writer knows the truth; the next read
   176→  // re-stats and re-populates (also covers EXTERNAL writers, which change
   177→  // the mtime/size and bust the cache naturally)
   178→  runsCache = null;
   179→}

   210→/**
   211→ * Non-null when a live child process is still tracked for this job —
   212→ * either through this server instance's `live` map, or via /proc on a
   213→ * record written before a hot reload / server restart. The run route
   214→ * uses this to refuse duplicate spawns (two mpirun trees writing to the
   215→ * same workdir = corrupt checkpoints + an OOM on this 4GB box).
   216→ * Returns a human-readable reason, or null when nothing is running.
   217→ */
   218→export function isRunAlive(jobId: string): string | null {
   219→  const child = live.get(jobId);
   220→  if (child && child.exitCode === null && child.pid != null && pidAlive(child.pid)) {
   221→    return `job is already running (pid ${child.pid})`;
   222→  }
   223→  const state = readRuns()[jobId];
   224→  if (state && state.done === false && state.pid != null && pidAlive(state.pid)) {
   225→    return `job is already running (pid ${state.pid})`;
   226→  }
   227→  return null;
   228→}

   230→/** All descendant pids of `pid` (mpirun → hydra → ranks), via /proc stat. */
   231→function descendantsOf(pid: number): number[] {
   232→  const ppidOf = new Map<number, number>();
   233→  try {
   234→    for (const entry of readdirSync("/proc")) {
   235→      if (!/^\d+$/.test(entry)) continue;
   236→      const p = Number(entry);
   237→      if (p === pid) continue;
   238→      try {
   239→        const stat = readFileSync(`/proc/${p}/stat`, "utf8");
   240→        const close = stat.lastIndexOf(")");
   241→        // fields after "comm": state, ppid, ... → ppid is rest[1]
   242→        const rest = stat.slice(close + 2).split(" ");
   243→        const ppid = Number(rest[1]);
   244→        if (Number.isFinite(ppid)) ppidOf.set(p, ppid);
   245→      } catch {
   246→        /* process vanished */
   247→      }
   248→    }
   249→  } catch {
   250→    return [];
   251→  }
   252→  const out: number[] = [];
   253→  const queue = [pid];
   254→  while (queue.length > 0) {
   255→    const parent = queue.shift() as number;
   256→    for (const [p, pp] of ppidOf) {
   257→      if (pp === parent && !out.includes(p)) {
   258→        out.push(p);
   259→        queue.push(p);
   260→      }
   261→    }
   262→  }
   263→  return out;
   264→}

   266→/** Cross-platform pid liveness: /proc on Linux, signal-0 probe on Windows. */
   267→function pidAlive(pid: number): boolean {
   268→  try {
   269→    if (existsSync("/proc")) return existsSync(`/proc/${pid}`);
   270→  } catch {
   271→    /* fall through to the signal probe */
   272→  }
   273→  try {
   274→    process.kill(pid, 0);
   275→    return true;
   276→  } catch (e) {
   277→    // EPERM = the process EXISTS but belongs to another session/user
   278→    // (e.g. a detached wsl.exe after a dev-server restart) — still alive.
   279→    // ESRCH (or anything else) = gone.
   280→    const err = e as NodeJS.ErrnoException;
   281→    return err.code === "EPERM";
   282→  }
   283→}

   285→/**
   286→ * Stop a job's live process tree (SIGTERM → grace → SIGKILL).
   287→ *  - kills mpirun AND its ranks (a bare mpirun kill orphans the ranks)
   288→ *  - if the child is in `live`, its exit handler fires and records
   289→    exit −1 → the next POST /run can --continue from the checkpoint
   290→ *  - for restart-orphaned trees (no live child, only a state record)
   291→    the record is marked interrupted directly
   292→ * Returns what happened, for the API response.
   293→ */
   294→export async function stopRun(jobId: string): Promise<{ stopped: boolean; message: string }> {
   295→  const child = live.get(jobId);
   296→  const state = readRuns()[jobId];
   297→  const pid = child?.pid ?? state?.pid ?? null;
   298→
   299→  // ---- WSL bridge runs --------------------------------------------------
   300→  // The host-side pid is the wsl.exe session host; the distro-internal
   301→  // mpirun/rank tree has no /proc entries on Windows. Kill the host session
   302→  // AND pkill the distro side by the translated workdir (always embedded in
   303→  // the RELION argv via --o / --i), so no orphan keeps writing checkpoints.
   304→  //
   305→  // BRIDGE-NESS COMES FROM THE RECORD, not from the live detection state:
   306→  // the wrapped display command starts with "wsl" / "wsl -d <distro> --" —
   307→  // relying on detectRelion() here made stopping impossible exactly when
   308→  // detection was stale/failed while the job kept running.
   309→  const bridged =
   310→    state?.cmd?.match(/^wsl(?: -d (\S+))? -- bash -c /) ?? null;
   311→  if (pid != null && state?.workdir && pidAlive(pid) && bridged) {
   312→    const distro = bridged[1] ?? null;
   313→    try {
   314→      process.kill(pid);
   315→    } catch {
   316→      /* already gone */
   317→    }
   318→    execFile("wsl.exe", wslStopArgs(state.workdir, distro), { timeout: 5000, windowsHide: true }, () => {
   319→      /* best effort — pkill exits non-zero when nothing matched */
   320→    });
   321→    if (!child) {
   322→      const runs = readRuns();
   323→      const rec = runs[jobId];
   324→      if (rec && rec.done === false) {
   325→        runs[jobId] = { ...rec, done: true, exitCode: -1, result: "stopped by user (WSL bridge)" };
   326→        writeRuns(runs);
   327→      }
   328→    }
   329→    return {
   330→      stopped: true,
   331→      message: `stopped WSL session pid ${pid} (pkill sent inside ${distro ?? "default distro"})`,
   332→    };
   333→  }
   334→
   335→  const tree = pid != null && existsSync(`/proc/${pid}`) ? [pid, ...descendantsOf(pid)] : [];
   336→
   337→  if (tree.length === 0) {
   338→    // Windows native fallback: no /proc tree exists, but the pid is alive
   339→    // (a non-bridged record on a win32 host). process.kill() terminates
   340→    // unconditionally there — no graceful SIGTERM, but the job stops.
   341→    if (pid != null && pidAlive(pid) && process.platform === "win32") {
   342→      try {
   343→        process.kill(pid);
   344→      } catch {
   345→        /* raced away */
   346→      }
   347→      return { stopped: true, message: `stopped pid ${pid} (Windows terminate)` };
   348→    }
   349→    return { stopped: false, message: "no live process for this job" };
   350→  }
   351→
   352→  for (const p of tree) {
   353→    try {
   354→      process.kill(p, "SIGTERM");
   355→    } catch {
   356→      /* already gone */
   357→    }
   358→  }
   359→  // graceful window: RELION ranks exit on SIGTERM, mpirun reaps them
   360→  const graceDeadline = Date.now() + 5000;
   361→  while (Date.now() < graceDeadline) {
   362→    await new Promise((r) => setTimeout(r, 250));
   363→    if (!tree.some((p) => pidAlive(p))) break;
   364→  }
   365→  let killed = 0;
   366→  for (const p of tree) {
   367→    if (pidAlive(p)) {
   368→      try {
   369→        process.kill(p, "SIGKILL");
   370→        killed += 1;
   371→      } catch {
   372→        /* raced away */
   373→      }
   374→    }
   375→  }
   376→
   377→  // If the exit handler won't fire (child not in `live` — e.g. the tree
   378→  // survived a server restart), mark the record interrupted ourselves so
   379→  // reconcile + the resume branch see a consistent state.
   380→  if (!child) {
   381→    const runs = readRuns();
   382→    const rec = runs[jobId];
   383→    if (rec && rec.done === false) {
   384→      runs[jobId] = { ...rec, done: true, exitCode: -1, result: "stopped by user" };
   385→      writeRuns(runs);
   386→    }
   387→  }
   388→  return {
   389→    stopped: true,
   390→    message: `stopped pid ${pid} (${tree.length} processes${killed > 0 ? `, ${killed} SIGKILLed` : ""})`,
   391→  };
   392→}

   398→/** Environment for spawning RELION binaries (PATH + libs + RELION_HOME). */
   399→export function relionEnv(binDir: string): NodeJS.ProcessEnv {
   400→  const pathParts: string[] = [binDir];
   401→  if (existsSync(MPICH_BIN)) pathParts.push(MPICH_BIN);
   402→  const libParts = [MPICH_LIB, FFTW_LIB].filter((p) => existsSync(p));
   403→  const env: NodeJS.ProcessEnv = {
   404→    ...process.env,
   405→    // path.delimiter (";" on Windows, ":" on POSIX) — a hardcoded ":" would
   406→    // corrupt PATH on native Windows runs
   407→    PATH: pathParts.join(path.delimiter) + path.delimiter + (process.env.PATH ?? ""),
   408→    RELION_HOME: binDir.replace(/\/bin\/?$/, ""),
   409→  };
   410→  // relion_run_ctffind does NOT search PATH — it needs an explicit executable
   411→  if (existsSync(CTFFIND_EXE)) env.RELION_CTFFIND_EXECUTABLE = CTFFIND_EXE;
   412→  // OpenMPI 4+ refuses mpirun as root (app itself running as root on a Linux
   413→  // host with an OpenMPI toolchain). Upstream-sanctioned opt-in pair; MPICH
   414→  // (sandbox) ignores them — harmless everywhere, inert for non-root users.
   415→  env.OMPI_ALLOW_RUN_AS_ROOT = "1";
   416→  env.OMPI_ALLOW_RUN_AS_ROOT_CONFIRM = "1";
   417→  env.LD_LIBRARY_PATH =
   418→    libParts.length > 0
   419→      ? libParts.join(":") + ":" + (process.env.LD_LIBRARY_PATH ?? "")
   420→      : (process.env.LD_LIBRARY_PATH ?? "");
   421→  return env;
   422→}

   463→/**
   464→ * Link a data directory into the RELION project/job tree so STAR files can
   465→ * use project-relative "micrographs/<name>" paths (RELION pipeliner style).
   466→ * POSIX hosts get a symlink; Windows gets a directory JUNCTION (no admin
   467→ * rights needed — plain symlinks require Developer Mode there). Returns
   468→ * false when no link could be created (e.g. junction to a UNC target).
   469→ */
   470→function linkDirInto(target: string, linkPath: string): boolean {
   471→  try {
   472→    if (existsSync(linkPath)) {
   473→      // a previous import may point elsewhere — re-point the link
   474→      try {
   475→        if (realpathSync(linkPath) === realpathSync(target)) return true;
   476→        rmSync(linkPath, { force: true, recursive: true });
   477→      } catch {
   478→        /* stale/broken link — remove and recreate below */
   479→        try {
   480→          rmSync(linkPath, { force: true, recursive: true });
   481→        } catch {
   482→          /* cannot clean up — keep the old link and fail honestly */
   483→          return false;
   484→        }
   485→      }
   486→    }
   487→    if (process.platform === "win32") {
   488→      symlinkSync(target, linkPath, "junction");
   489→    } else {
   490→      symlinkSync(target, linkPath, "dir");
   491→    }
   492→    return true;
   493→  } catch {
   494→    return false;
   495→  }
   496→}

   498→/**
   499→ * De-duplicate a STAR entry name for file-set imports. Collisions get the
   500→ * parent folder as a prefix (MovieDir__frame.mrc), then -2, -3… suffixes.
   501→ */
   502→function uniqueStarName(file: string, used: Set<string>): string {
   503→  let name = path.basename(file);
   504→  if (used.has(name)) {
   505→    const ext = path.extname(name);
   506→    const stem = name.slice(0, name.length - ext.length);
   507→    const parent = path.basename(path.dirname(file)) || "src";
   508→    name = `${parent}__${stem}${ext}`;
   509→    let i = 2;
   510→    while (used.has(name)) {
   511→      name = `${parent}__${stem}-${i}${ext}`;
   512→      i += 1;
   513→    }
   514→  }
   515→  used.add(name);
   516→  return name;
   517→}

   519→/**
   520→ * Import a SET of files (multi-select / wildcard pattern / single file):
   521→ * create a real "micrographs" directory in the project tree and link every
   522→ * file into it — hardlink first (no data duplication, works on POSIX and
   523→ * NTFS), then a file symlink, then absolute STAR paths as honest fallback
   524→ * (translated to WSL-side paths when the bridge is active).
   525→ * Unlinkable files additionally get a .pathref marker so the gallery/preview
   526→ * routes can still serve them from their source location — without the
   527→ * marker an absolute-path import LOOKS like "nothing was imported" even
   528→ * though the STAR file references every file correctly.
   529→ * Returns the STAR entries + how many files could NOT be linked.
   530→ */
   531→function importFileSet(
   532→  hostFiles: string[],
   533→  projectDir: string,
   534→  workdir: string,
   535→  bridge: WslBridge | null
   536→): { entries: string[]; unlinked: number } {
   537→  const projectMic = path.join(projectDir, "micrographs");
   538→  const workdirMic = path.join(workdir, "micrographs");
   539→  // fresh real dir — removes a previous import's symlink/junction (the link
   540→  // only, never the target data) or an older file-set's hardlinks
   541→  for (const stale of [workdirMic, projectMic]) {
   542→    try {
   543→      rmSync(stale, { force: true, recursive: true });
   544→    } catch {
   545→      /* non-fatal: linkDirInto below re-points what it can */
   546→    }
   547→  }
   548→  mkdirSync(projectMic, { recursive: true });
   549→
   550→  const used = new Set<string>();
   551→  const entries: string[] = [];
   552→  /** unlinkable files: name → source, for pathref markers */
   553→  const unlinkedFiles: { name: string; abs: string }[] = [];
   554→  for (const f of hostFiles) {
   555→    const name = uniqueStarName(f, used);
   556→    const target = path.join(projectMic, name);
   557→    let linked = false;
   558→    try {
   559→      linkSync(f, target); // hardlink — same filesystem only
   560→      linked = true;
   561→    } catch {
   562→      try {
   563→        symlinkSync(f, target, "file");
   564→        linked = true;
   565→      } catch {
   566→        linked = false; // cross-volume + no symlink rights → absolute path
   567→      }
   568→    }
   569→    if (linked) {
   570→      entries.push(`micrographs/${name}`);
   571→    } else {
   572→      unlinkedFiles.push({ name, abs: f });
   573→      entries.push(bridge ? hostToWsl(f) : f);
   574→    }
   575→  }
   576→  // expose the set inside the import job's workdir (Files tab + gallery)
   577→  if (!linkDirInto(projectMic, workdirMic)) {
   578→    // junction failed — link each file directly into the workdir instead
   579→    for (const e of entries) {
   580→      if (!e.startsWith("micrographs/")) continue;
   581→      const src = path.join(projectMic, e.slice("micrographs/".length));
   582→      const dst = path.join(workdirMic, e.slice("micrographs/".length));
   583→      try {
   584→        mkdirSync(path.dirname(dst), { recursive: true });
   585→        linkSync(src, dst);
   586→      } catch {
   587→        try {
   588→          mkdirSync(path.dirname(dst), { recursive: true });
   589→          symlinkSync(src, dst, "file");
   590→        } catch {
   591→          /* gallery preview lost for this file — import still works */
   592→        }
   593→      }
   594→    }
   595→  }
   596→  // markers for the unlinkable files (both trees — the junction case only
   597→  // carries projectMic's markers into the workdir automatically)
   598→  for (const u of unlinkedFiles) {
   599→    writePathrefMarker(projectMic, u.name, u.abs);
   600→    try {
   601→      mkdirSync(workdirMic, { recursive: true });
   602→      writePathrefMarker(workdirMic, u.name, u.abs);
   603→    } catch {
   604→      /* workdir marker is best-effort */
   605→    }
   606→  }
   607→  return { entries, unlinked: unlinkedFiles.length };
   608→}

   629→/** Micrograph pixel size from the pipeline's Import job (Å). */
   630→function micAngpix(upstream: UpstreamRef[]): number | null {
   631→  const importUp = upstream.find((u) => u.type === "import");
   632→  return importUp?.params && typeof importUp.params.pixelSize === "number"
   633→    ? importUp.params.pixelSize
   634→    : null;

   637→/** Particle pixel size: import pixel × (extract box / downsample). */
   638→function particlePixel(job: EngineJobRef, upstream: UpstreamRef[]): number {
   639→  const importUp = upstream.find((u) => u.type === "import" || u.type === "tomo_import");
   640→  const extractUp = upstream.find((u) => u.type === "extract");
   641→  const pixel =
   642→    importUp?.params && typeof importUp.params.pixelSize === "number"
   643→      ? importUp.params.pixelSize
   644→      : 1.77;
   645→  if (extractUp?.params) {
   646→    const box = typeof extractUp.params.boxSize === "number" ? extractUp.params.boxSize : 128;
   647→    const down =
   648→      typeof extractUp.params.downsampleTo === "number" ? extractUp.params.downsampleTo : 0;
   649→    if (down > 0 && down < box) return (pixel * box) / down;
   650→  }
   651→  return pixel;
   652→}

   816→/**
   817→ * Resolve required inputs from upstream completed runs.
   818→ * `missing` is a human-readable message (null on success); `wait` classifies
   819→ * WHY an input is missing so the dispatcher can mark the job PENDING
   820→ * (upstream failed / still running / never ran) instead of failed — a failed
   821→ * upstream used to cascade red "Waiting for upstream output" failures down
   822→ * the entire pipeline.
   823→ */
   824→function resolveInputs(
   825→  type: string,
   826→  upstream: UpstreamRef[],
   827→  params?: Record<string, unknown>
   828→): { inputs: Record<string, string>; missing: string | null; wait?: WaitKind } {
   829→  // param-driven optionality: drop requirements whose skipIf fires
   830→  // (only the dispatch call passes params — the manualpick/select
   831→  // pre-flight lookups have no skipIf requirements).
   832→  const reqs = (INPUTS[type] ?? []).filter(
   833→    (r) => !r.skipIf || !params || !r.skipIf(params)
   834→  );
   835→  const runs = readRuns();
   836→  const inputs: Record<string, string> = {};
   837→
   838→  for (const req of reqs) {
   839→    let resolved: string | null = null;
   840→    // upstream arrives in INPUT-PRIORITY order from lineageFor(): BFS by
   841→    // graph distance (direct parents first) with newest-first within a
   842→    // layer — scan forward and take the first provider that has the output.
   843→    const providers: { name: string; status: string }[] = [];
   844→    for (const up of upstream) {
   845→      if (!req.from.includes(up.type)) continue;
   846→      const state = runs[up.id];
   847→      if (state && state.done && state.exitCode === 0) {
   848→        for (const key of req.accepts) {
   849→          const p = state.outputs[key];
   850→          if (p && existsSync(p)) {
   851→            resolved = p;
   852→            break;
   853→          }
   854→        }
   855→        if (resolved) break;
   856→      }
   857→      providers.push({ name: up.name ?? up.type, status: up.status ?? "idle" });
   858→    }
   859→    if (!resolved) {
   860→      // optional inputs degrade silently (gallery sources, previews)
   861→      if (req.optional) continue;
   862→      // pick the most actionable blocker: a FAILED provider (re-run it)
   863→      // beats a RUNNING one (transient wait) beats anything else
   864→      const failed = providers.find((p) => p.status === "failed");
   865→      const running = providers.find((p) => p.status === "running");
   866→      if (failed) {
   867→        return {
   868→          inputs: {},
   869→          missing: `Upstream "${failed.name}" failed — fix and re-run it; this job then starts automatically once its inputs are ready`,
   870→          wait: "upstream-failed",
   871→        };
   872→      }
   873→      if (running) {
   874→        return {
   875→          inputs: {},
   876→          missing: `Waiting for upstream "${running.name}" to finish… (this job starts automatically when it does)`,
   877→          wait: "upstream-running",
   878→        };
   879→      }
   880→      return {
   881→        inputs: {},
   882→        missing: `Waiting for upstream output: ${req.label} — runs automatically once ready`,
   883→        wait: "not-ready",
   884→      };
   885→    }
   886→    inputs[req.key] = resolved;
   887→  }
   888→  return { inputs, missing: null };
   889→}

   937→/* ------------------------------------------------------------------ */
   938→/* STAR file helpers                                                    */
   939→/* ------------------------------------------------------------------ */
   940→
   941→interface StarBlock {
   942→  /** "data_xxx" header line. */
   943→  header: string;
   944→  /** All following lines (including loop_/labels/rows). */
   945→  lines: string[];
   946→}
   947→
   948→function parseStarBlocks(text: string): StarBlock[] {
   949→  const blocks: StarBlock[] = [];
   950→  let current: StarBlock | null = null;
   951→  for (const raw of text.split(/\r?\n/)) {
   952→    const line = raw.trimEnd();
   953→    if (/^data_/.test(line.trim())) {
   954→      current = { header: line.trim(), lines: [] };
   955→      blocks.push(current);
   956→    } else if (current) {
   957→      current.lines.push(line);
   958→    }
   959→  }
   960→  return blocks;
   961→}
   962→
   963→/** Micrograph names from a micrographs.star (first column after the loop header). */
   964→function micrographNames(starPath: string): string[] {
   965→  const blocks = parseStarBlocks(readFileSync(starPath, "utf8"));
   966→  const micBlock = blocks.find(
   967→    (b) => b.header === "data_micrographs" || b.lines.some((l) => l.includes("_rlnMicrographName"))
   968→  );
   969→  if (!micBlock) return [];
   970→  const names: string[] = [];
   971→  for (const line of micBlock.lines) {
   972→    const t = line.trim();
   973→    if (!t || t.startsWith("#") || t === "loop_" || t.startsWith("_rln") || t.startsWith("data_")) {
   974→      continue;
   975→    }
   976→    const first = t.split(/\s+/)[0];
   977→    if (first) names.push(first);
   978→  }
   979→  return names;
   980→}

  1025→/** Import: writes a RELION 5 optics-group micrographs.star (EMPIAR or empty). */
  1026→async function runImportNative(job: EngineJobRef): Promise<NativeResult> {
  1027→  const workdir = workdirFor(job);
  1028→  mkdirSync(workdir, { recursive: true });
  1029→  const projectDir = projectDirFor(job);
  1030→  mkdirSync(projectDir, { recursive: true });
  1031→
  1032→  const pixel = num(job, "pixelSize", 1.77);
  1033→  const kV = num(job, "voltage", 300);
  1034→  const cs = num(job, "cs", 2.7);
  1035→  const q0 = num(job, "ampContrast", 0.1);
  1036→  const empiar =
  1037→    String(job.params.empiarData ?? "") === "true" &&
  1038→    !String(job.params.micrographsPath ?? "").trim();
  1039→
  1040→  const starPath = path.join(workdir, "micrographs.star");
  1041→  const lines: string[] = [
  1042→    "data_optics",
  1043→    "",
  1044→    "loop_",
  1045→    "_rlnOpticsGroup #1",
  1046→    "_rlnOpticsGroupName #2",
  1047→    "_rlnMicrographPixelSize #3",
  1048→    "_rlnVoltage #4",
  1049→    "_rlnSphericalAberration #5",
  1050→    "_rlnAmplitudeContrast #6",
  1051→    `1 optGroup1 ${pixel} ${kV} ${cs} ${q0}`,
  1052→    "",
  1053→    "data_micrographs",
  1054→    "",
  1055→    "loop_",
  1056→    "_rlnMicrographName #1",
  1057→    "_rlnOpticsGroup #2",
  1058→  ];
  1059→
  1060→  let result: string;
  1061→  let sourceLabel = "";
  1062→  if (empiar) {
  1063→    const mrcs = existsSync(EMPIAR_DIR)
  1064→      ? readdirSync(EMPIAR_DIR)
  1065→          .filter((f) => f.endsWith(".mrc"))
  1066→          .sort()
  1067→      : [];
  1068→    if (mrcs.length === 0) {
  1069→      return { ok: false, error: `EMPIAR directory not found: ${EMPIAR_DIR}` };
  1070→    }
  1071→    // Project-relative paths + a symlink so every downstream RELION job (run
  1072→    // with CWD = projectDir) can open "micrographs/<name>.mrc". This mirrors
  1073→    // the RELION pipeliner: STAR paths are project-root-relative.
  1074→    const micLink = path.join(projectDir, "micrographs");
  1075→    if (!existsSync(micLink)) symlinkSync(EMPIAR_DIR, micLink);
  1076→    // Also expose the micrographs inside the import job's own workdir so the
  1077→    // Files tab + gallery can serve PNG previews through outputs/file.
  1078→    const micLinkInWorkdir = path.join(workdir, "micrographs");
  1079→    if (!existsSync(micLinkInWorkdir)) symlinkSync(EMPIAR_DIR, micLinkInWorkdir);
  1080→    for (const m of mrcs) lines.push(`micrographs/${m} 1`);
  1081→    result = `${mrcs.length} micrographs imported · EMPIAR-10017 (pixel ${pixel} Å)`;
  1082→    sourceLabel = `source: EMPIAR-10017 ${EMPIAR_DIR}`;
  1083→  } else {
  1084→    // ---- user-selected source — RELION "Select files by" forms ----------
  1085→    // micrographsPath accepts THREE shapes (exactly one param):
  1086→    //   1. a folder            → import every image inside it
  1087→    //   2. a wildcard pattern  → /data/movies/*.tiff — expanded here
  1088→    //   3. file paths          → newline-separated multi-select list
  1089→    const customRaw = String(job.params.micrographsPath ?? "").trim();
  1090→    if (customRaw) {
  1091→      const status = await detectRelion();
  1092→      const bridge = bridgeFromStatus(status);
  1093→      const toHost = (p: string) => userPathToHost(p, bridge?.distro ?? null);
  1094→
  1095→      const listed = customRaw.split(/\r?\n/).map((s) => s.trim()).filter(Boolean);
  1096→      const multiFile = listed.length > 1;
  1097→      const single = listed[0] ?? "";
  1098→      const isPattern = !multiFile && hasWildcard(single);
  1099→
  1100→      let hostFiles: string[] = [];
  1101→      let skipped = 0; // non-image files filtered out of an explicit set
  1102→
  1103→      if (multiFile) {
  1104→        // ---- 3. explicit multi-select file list ---------------------------
  1105→        for (const line of listed) {
  1106→          const host = toHost(line);
  1107→          try {
  1108→            if (!statSync(host).isFile()) {
  1109→              return {
  1110→                ok: false,
  1111→                error: `Not a file: ${line} — re-pick the micrographs in the params tab (Browse → Files)`,
  1112→              };
  1113→            }
  1114→          } catch {
  1115→            return {
  1116→              ok: false,
  1117→              error: `File not accessible: ${line} — re-pick the micrographs in the params tab (Browse → Files)`,
  1118→            };
  1119→          }
  1120→          if (MIC_RE.test(path.basename(host))) hostFiles.push(host);
  1121→          else skipped += 1;
  1122→        }
  1123→      } else if (isPattern) {
  1124→        // ---- 2. wildcard pattern (RELION "File name pattern") -------------
  1125→        const expanded = expandPattern(single, toHost);
  1126→        if ("error" in expanded) {
  1127→          return { ok: false, error: `${expanded.error} — check the pattern in the params tab` };
  1128→        }
  1129→        for (const f of expanded.files) {
  1130→          if (MIC_RE.test(path.basename(f))) hostFiles.push(f);
  1131→          else skipped += 1;
  1132→        }
  1133→        if (expanded.total > expanded.files.length) {
  1134→          sourceLabel = `source: pattern ${single} (${expanded.total} matches, first ${expanded.files.length} imported)`;
  1135→        }
  1136→      } else {
  1137→        // ---- 1. folder (or a single pasted file) --------------------------
  1138→        const hostDir = toHost(single);
  1139→        let st: { isDirectory: boolean; isFile: boolean } | null = null;
  1140→        try {
  1141→          const s = statSync(hostDir);
  1142→          st = { isDirectory: s.isDirectory(), isFile: s.isFile() };
  1143→        } catch {
  1144→          st = null;
  1145→        }
  1146→        if (!st) {
  1147→          return {
  1148→            ok: false,
  1149→            error: `Micrographs folder not accessible: ${customRaw} — re-pick it in the params tab (Browse…)`,
  1150→          };
  1151→        }
  1152→        if (st.isDirectory) {
  1153→          const mrcs = readdirSync(hostDir)
  1154→            .filter((f) => MIC_RE.test(f))
  1155→            .sort();
  1156→          if (mrcs.length === 0) {
  1157→            return {
  1158→              ok: false,
  1159→              error: `No .mrc/.mrcs/.tif/.eer micrographs found in ${customRaw}`,
  1160→            };
  1161→          }
  1162→          hostFiles = mrcs.map((m) => path.join(hostDir, m));
  1163→        } else {
  1164→          if (!MIC_RE.test(path.basename(hostDir))) {
  1165→            return {
  1166→              ok: false,
  1167→              error: `Not a micrograph file (.mrc/.mrcs/.tif/.tiff/.eer): ${customRaw}`,
  1168→            };
  1169→          }
  1170→          hostFiles = [hostDir];
  1171→        }
  1172→      }
  1173→
  1174→      if (hostFiles.length === 0) {
  1175→        return {
  1176→          ok: false,
  1177→          error: multiFile
  1178→            ? `No image files (.mrc/.mrcs/.tif/.tiff/.eer) among the ${listed.length} selected paths`
  1179→            : `No image files (.mrc/.mrcs/.tif/.tiff/.eer) match the pattern ${single}`,
  1180→        };
  1181→      }
  1182→
  1183→      let unlinked = 0;
  1184→      if (!multiFile && !isPattern) {
  1185→        // folder import — link the WHOLE directory (one junction/symlink,
  1186→        // star stays project-relative; transparent to WSL drvfs too)
  1187→        const hostDir = toHost(single);
  1188→        const linked =
  1189→          linkDirInto(hostDir, path.join(projectDir, "micrographs")) &&
  1190→          linkDirInto(hostDir, path.join(workdir, "micrographs"));
  1191→        if (linked) {
  1192→          for (const f of hostFiles) lines.push(`micrographs/${path.basename(f)} 1`);
  1193→        } else {
  1194→          // no link possible (e.g. junction to a UNC \\wsl.localhost target) —
  1195→          // absolute paths: RELION opens them as-is; on a Windows host with the
  1196→          // bridge they are translated to WSL-side /… paths.
  1197→          for (const f of hostFiles) {
  1198→            lines.push(`${bridge ? hostToWsl(f) : f} 1`);
  1199→            unlinked += 1;
  1200→          }
  1201→        }
  1202→      } else {
  1203→        // pattern / file-list — link the FILES into a real project dir
  1204→        const r = importFileSet(hostFiles, projectDir, workdir, bridge);
  1205→        for (const e of r.entries) lines.push(`${e} 1`);
  1206→        unlinked = r.unlinked;
  1207→      }
  1208→
  1209→      const skipNote = skipped > 0 ? ` · ${skipped} non-image file${skipped === 1 ? "" : "s"} skipped` : "";
  1210→      const linkedCount = hostFiles.length - unlinked;
  1211→      const linkNote =
  1212→        unlinked > 0
  1213→          ? ` · ${linkedCount} linked, ${unlinked} referenced by path (source not linkable)`
  1214→          : "";
  1215→      const kindNote = isPattern ? " · pattern" : multiFile ? " · file list" : "";
  1216→      result = `${hostFiles.length} micrographs imported${kindNote}${skipNote}${linkNote} (pixel ${pixel} Å)`;
  1217→      if (!sourceLabel) {
  1218→        sourceLabel =
  1219→          "source: " +
  1220→          (isPattern ? "pattern " : multiFile ? `${listed.length} selected files — ` : "") +
  1221→          customRaw.slice(0, 200) +
  1222→          (unlinked > 0 ? " (absolute paths)" : "");
  1223→      }
  1224→    } else {
  1225→      result = "Import job completed (no source data configured — pick a micrographs folder, pattern or files in the params tab)";
  1226→      sourceLabel = "source: (none configured)";
  1227→    }
  1228→  }
  1229→
  1230→  writeFileSync(starPath, lines.join("\n") + "\n");
  1231→  const logText = [
  1232→    `CryoFlow engine-native import ${new Date().toISOString()}`,
  1233→    `pixel=${pixel} Å  voltage=${kV} kV  Cs=${cs} mm  Q0=${q0}`,
  1234→    sourceLabel,
  1235→    `output: ${starPath}`,
  1236→    result,
  1237→    "",
  1238→  ].join("\n");
  1239→  recordNativeRun(
  1240→    job,
  1241→    workdir,
  1242→    "engine-native: write micrographs.star (RELION 5 optics format)",
  1243→    { micrographs_star: starPath },
  1244→    result,
  1245→    logText
  1246→  );
  1247→  return { ok: true, result };
  1248→}

  1250→/** ManualPick: import Henderson .coord files as a RELION pick STAR. */
  1251→async function runManualPickNative(job: EngineJobRef, upstream: UpstreamRef[]): Promise<NativeResult> {
  1252→  const resolved = resolveInputs("manualpick", upstream);
  1253→  if (resolved.missing) return { ok: false, error: resolved.missing, wait: resolved.wait };
  1254→  const micStar = resolved.inputs.micrographs_star;
  1255→
  1256→  const workdir = workdirFor(job);
  1257→  mkdirSync(workdir, { recursive: true });
  1258→  const starPath = path.join(workdir, "manualpick.star");
  1259→
  1260→  const mics = micrographNames(micStar);
  1261→  if (mics.length === 0) {
  1262→    return { ok: false, error: `No micrographs found in ${micStar} (run Import first)` };
  1263→  }
  1264→  const projectDir = projectDirFor(job);
  1265→  // Star paths are project-root-relative — resolve to absolute for FS access.
  1266→  // Absolute entries (import's no-link fallback) may be WSL-side paths on a
  1267→  // Windows host → translate /mnt/<drive> and distro paths back to host form.
  1268→  const bridge = bridgeFromStatus(await detectRelion());
  1269→  const toAbs = (mic: string) => {
  1270→    const abs = mic.startsWith("/") ? mic : path.join(projectDir, mic);
  1271→    if (!isWindowsPath(abs) && abs.startsWith("/") && !existsSync(abs)) {
  1272→      // WSL-side path this host cannot open as-is
  1273→      const mnt = abs.match(/^\/mnt\/([A-Za-z])\/(.*)$/);
  1274→      if (mnt) return `${mnt[1].toUpperCase()}:\\${mnt[2].replace(/\//g, "\\")}`;
  1275→      if (bridge?.distro) {
  1276→        return `\\\\wsl.localhost\\${bridge.distro}\\${abs.slice(1).replace(/\//g, "\\")}`;
  1277→      }
  1278→    }
  1279→    return abs;
  1280→  };
  1281→
  1282→  const lines: string[] = [
  1283→    "data_particles",
  1284→    "",
  1285→    "loop_",
  1286→    "_rlnCoordinateX #1",
  1287→    "_rlnCoordinateY #2",
  1288→    "_rlnMicrographName #3",
  1289→  ];
  1290→  let total = 0;
  1291→  const missing: string[] = [];
  1292→  // Also write plain .coord files (RELION extract consumes them via
  1293→  // --coord_dir/--coord_suffix) — readCoordinates() parses "x y" ASCII.
  1294→  const coordsDir = path.join(workdir, "micrographs");
  1295→  mkdirSync(coordsDir, { recursive: true });
  1296→  const coordText: string[] = [];
  1297→  for (const mic of mics) {
  1298→    const micAbs = toAbs(mic);
  1299→    const coord = micAbs.replace(/\.mrcs?$/i, "") + ".coord";
  1300→    if (!existsSync(coord)) {
  1301→      missing.push(path.basename(coord));
  1302→      continue;
  1303→    }
  1304→    const rows = readFileSync(coord, "utf8")
  1305→      .split(/\r?\n/)
  1306→      .map((l) => l.trim())
  1307→      .filter((l) => l.length > 0 && !l.startsWith("#"));
  1308→    const outRows: string[] = [];
  1309→    for (const row of rows) {
  1310→      const parts = row.split(/\s+/);
  1311→      if (parts.length >= 2) {
  1312→        // keep the project-relative micrograph name (matches import star)
  1313→        lines.push(`${parts[0]} ${parts[1]} ${mic}`);
  1314→        outRows.push(`${parts[0]} ${parts[1]}`);
  1315→        total++;
  1316→      }
  1317→    }
  1318→    if (outRows.length > 0) {
  1319→      const base = path.basename(mic).replace(/\.mrcs?$/i, "");
  1320→      writeFileSync(path.join(coordsDir, base + ".coord"), outRows.join("\n") + "\n");
  1321→      coordText.push(`  ${base}.coord (${outRows.length} picks)`);
  1322→    }
  1323→    // expose the picked micrograph frames next to their coord files so the
  1324→    // picks preview panel can render overlay thumbnails via outputs/file
  1325→    const micLink = path.join(coordsDir, path.basename(mic));
  1326→    if (!existsSync(micLink)) {
  1327→      try {
  1328→        symlinkSync(micAbs, micLink);
  1329→      } catch {
  1330→        /* best-effort — preview only */
  1331→      }
  1332→    }
  1333→  }
  1334→  if (total === 0) {
  1335→    return {
  1336→      ok: false,
  1337→      error: `No .coord pick files found beside the micrographs (expected Henderson picks in ${EMPIAR_DIR}) — missing: ${missing.join(", ")}`,
  1338→    };
  1339→  }
  1340→
  1341→  writeFileSync(starPath, lines.join("\n") + "\n");
  1342→  const result = `${total} picks imported from Henderson .coord files`;
  1343→  const logText = [
  1344→    `CryoFlow engine-native manualpick ${new Date().toISOString()}`,
  1345→    `micrographs: ${mics.length} (from ${micStar})`,

  1363→/** Select: class-aware or first-N subset of upstream particles.star (engine-native).
  1364→ *

  1366→ * Class2D/Class3D job) and classCutoff > 0, classes whose occupancy is below
  1367→ * cutoff × (largest class occupancy) are pruned — the programmatic equivalent
  1368→ * of RELION's "select good 2D classes by occupancy" workflow.
  1369→ * Otherwise: first-N subset.

  1380→  const text = readFileSync(inStar, "utf8");
  1381→  const blocks = parseStarBlocks(text);
  1382→  const outLines: string[] = [];
  1383→  let total = 0;
  1384→  let kept = 0;

  1386→  const cutoff = Math.max(0, Math.min(1, num(job, "classCutoff", 0.5)));
  1387→  const classStats: { cls: string; count: number; kept: boolean }[] = [];
  1388→  let keptClasses = 0;
  1389→
  1390→  for (const block of blocks) {
  1391→    outLines.push(block.header, "");
  1392→    const lines = block.lines;
  1393→    // find the loop header boundary
  1394→    let i = 0;
  1395→    const isParticles = lines.some((l) => l.trim().startsWith("_rlnImageName"));
  1396→    if (!isParticles) {
  1397→      for (const l of lines) outLines.push(l);
  1398→      outLines.push("");
  1399→      continue;
  1400→    }
  1401→    // header lines until the first data row
  1402→    while (
  1403→      i < lines.length &&
  1404→      (lines[i].trim() === "" ||
  1405→        lines[i].trim() === "loop_" ||
  1406→        lines[i].trim().startsWith("_rln") ||
  1407→        lines[i].trim().startsWith("#"))

  1414→    const classCol = labelColumn(lines, i, "_rlnClassNumber");

  1425→        if (Number.isFinite(cls)) counts.set(cls, (counts.get(cls) ?? 0) + 1);
  1426→      }
  1427→      const maxCount = Math.max(0, ...counts.values());
  1428→      keepClasses = new Set<number>();
  1429→      for (const [cls, count] of [...counts.entries()].sort((a, b) => b[1] - a[1])) {
  1430→        const ok = count >= cutoff * maxCount;
  1431→        classStats.push({ cls: String(cls), count, kept: ok });
  1432→        if (ok) {
  1433→          keepClasses.add(cls);
  1434→          keptClasses++;
  1435→        }
  1436→      }
  1437→    }
  1438→
  1439→    for (let r = i; r < lines.length; r++) {
  1440→      const t = lines[r].trim();

  1442→      total++;
  1443→      let keep = kept < maxN;
  1444→      if (keep && keepClasses) {
  1445→        const cells = t.split(/\s+/);
  1446→        const cls = parseInt(cells[classCol] ?? "", 10);
  1447→        keep = Number.isFinite(cls) && keepClasses.has(cls);
  1448→      }
  1449→      if (keep) {
  1450→        outLines.push(t);
  1451→        kept++;
  1452→      }
  1453→    }
  1454→    outLines.push("");
  1455→  }
  1456→
  1457→  writeFileSync(outStar, outLines.join("\n") + "\n");

  1459→    classStats.length > 0
  1460→      ? `${kept} of ${total} particles selected · kept ${keptClasses}/${classStats.length} classes (occupancy ≥ ${cutoff}× best)`
  1461→      : `${kept} of ${total} particles selected`;
  1462→  const logText = [
  1463→    `CryoFlow engine-native select ${new Date().toISOString()}`,
  1464→    `input:  ${inStar} (${total} particles)`,
  1465→    ...(classStats.length > 0
  1466→      ? [
  1467→          "class occupancy (count · kept):",
  1468→          ...classStats
  1469→            .sort((a, b) => b.count - a.count)
  1470→            .map((c) => `  class ${c.cls}: ${c.count} · ${c.kept ? "kept" : "PRUNED"}`),
  1471→        ]
  1472→      : [`mode: first-N (no _rlnClassNumber column or classCutoff=0)`]),
  1473→    `output: ${outStar} (${kept} particles)`,
  1474→    result,
  1475→    "",
  1476→  ].join("\n");
  1477→  recordNativeRun(job, workdir, "engine-native: particle selection (class-aware)", { particles_star: outStar }, result, logText);
  1478→  return { ok: true, result };
  1479→}

  1623→/** Symmetry Expansion: replicate every particle row |G|× with composed ZYZ

  1659→  let expanded = 0;

  1661→  for (const block of blocks) {
  1662→    outLines.push(block.header, "");
  1663→    const lines = block.lines;
  1664→    const isParticles = lines.some((l) => l.trim().startsWith("_rlnImageName"));
  1665→    if (!isParticles) {
  1666→      for (const l of lines) outLines.push(l);

  1672→      i < lines.length &&
  1673→      (lines[i].trim() === "" ||
  1674→        lines[i].trim() === "loop_" ||

  1739→    job,
  1740→    workdir,

  1922→/**
  1923→ * Resolve an external program executable.
  1924→ *  - native installs: bin dir first, then the host PATH (`which`)
  1925→ *  - WSL-bridged installs: the bin dir and PATH live INSIDE the distro —
  1926→ *    host existsSync/`which` cannot see them (and `which` does not exist on
  1927→ *    Windows at all) — so ask the distro itself with `command -v`.
  1928→ */
  1929→async function externalOnPath(
  1930→  binDir: string,
  1931→  names: string[],
  1932→  bridge: WslBridge | null
  1933→): Promise<string | null> {
  1934→  for (const n of names) {
  1935→    if (bridge) {
  1936→      // distro-side lookup — one wsl.exe call per name
  1937→      try {
  1938→        const args: string[] = [];
  1939→        if (bridge.distro) args.push("-d", bridge.distro);
  1940→        const q = binDir.replace(/'/g, "'\\''");
  1941→        const nm = n.replace(/'/g, "'\\''");
  1942→        args.push(
  1943→          "-e", "bash", "-lc",
  1944→          `command -v '${nm}' 2>/dev/null || { test -x '${q}'/'${nm}' && printf '%s' '${q}/${n}'; } || true`
  1945→        );
  1946→        const found = await new Promise<string>((resolve) => {
  1947→          execFile("wsl.exe", args, { timeout: 4000, windowsHide: true }, (err, stdout) => {
  1948→            const out = String(stdout ?? "").trim();
  1949→            resolve(!err && out.startsWith("/") ? out : "");
  1950→          });
  1951→        });
  1952→        if (found) return found;
  1953→      } catch {
  1954→        /* distro unreachable — the honest not-found error below */
  1955→      }
  1956→    } else {
  1957→      if (existsSync(binJoin(binDir, n))) return binJoin(binDir, n);
  1958→      const found = await new Promise<string | null>((resolve) => {
  1959→        execFile("which", [n], { timeout: 2000, windowsHide: true }, (err, stdout) => {
  1960→          const out = String(stdout ?? "").trim();
  1961→          resolve(!err && out.includes("/") ? out : null);
  1962→        });
  1963→      });
  1964→      if (found) return found;
  1965→    }
  1966→  }
  1967→  return null;
  1968→}
