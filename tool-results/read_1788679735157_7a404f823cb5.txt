  1209→export function jobType(key: string): JobTypeSpec | undefined {
  1210→  return TYPE_MAP.get(key);
  1211→}

  1213→/** Default parameter map for a job type (used by seeding + forms). */
  1214→export function defaultParams(key: string): Record<string, ParamValue> {
  1215→  const t = jobType(key);
  1216→  if (!t) return {};
  1217→  const out: Record<string, ParamValue> = {};
  1218→  for (const p of t.params) out[p.key] = p.default;
  1219→  return out;
  1220→}
  1221→
  1222→/** All tabs declared by the spec + any param tabs, in order, deduped. */
  1223→export function tabsFor(t: JobTypeSpec | undefined): string[] {
  1224→  if (!t) return [];
  1225→  const tabs: string[] = [];
  1226→  for (const name of [...t.tabs, ...t.params.map((p) => p.tab ?? "")]) {
  1227→    if (name && !tabs.includes(name)) tabs.push(name);
  1228→  }
  1229→  return tabs.length > 0 ? tabs : ["Parameters"];
  1230→}

  1236→/** Y coordinate of the i-th port (of count) on a card edge. */
  1237→export function portY(i: number, count: number): number {
  1238→  if (count <= 1) return CARD_H / 2;
  1239→  return (CARD_H * (i + 1)) / (count + 1);
  1240→}
  1241→
  1242→/** Can the output port of one job type feed the input port of another? */
  1243→export function portsCompatible(
  1244→  fromType: string,
  1245→  fromPort: string,
  1246→  toType: string,
  1247→  toPort: string
  1248→): boolean {
  1249→  const from = jobType(fromType);
  1250→  const to = jobType(toType);
  1251→  const o = from?.outputs.find((p) => p.name === fromPort);
  1252→  const i = to?.inputs.find((p) => p.name === toPort);
  1253→  if (!o || !i) return false;
  1254→  const accepts = i.accepts ?? ["*"];
  1255→  return accepts.includes("*") || (o.kind != null && accepts.includes(o.kind));
  1256→}
  1257→
  1258→/**
  1259→ * Default (first compatible) output→input port pair between two job types.
  1260→ * Used to give legacy DB edges a sensible port mapping.
  1261→ */
  1262→export function defaultPorts(
  1263→  fromType: string,
  1264→  toType: string
  1265→): { fromPort?: string; toPort?: string } {
  1266→  const from = jobType(fromType);
  1267→  const to = jobType(toType);
  1268→  if (!from || !to) return {};
  1269→  for (const o of from.outputs) {
  1270→    for (const i of to.inputs) {
  1271→      const accepts = i.accepts ?? ["*"];
  1272→      if (accepts.includes("*") || (o.kind != null && accepts.includes(o.kind))) {
  1273→        return { fromPort: o.name, toPort: i.name };
  1274→      }
  1275→    }
  1276→  }
  1277→  return {
  1278→    fromPort: from.outputs[0]?.name,
  1279→    toPort: to.inputs[0]?.name,
  1280→  };
  1281→}

  1287→/**
  1288→ * Coerce a stored raw value into the schema-typed value. Handles legacy
  1289→ * "true"/"false" strings for bool params and stringified numbers.
  1290→ */
  1291→export function coerceParam(p: ParamSchema, raw: unknown): ParamValue {
  1292→  if (p.type === "bool") {
  1293→    if (typeof raw === "boolean") return raw;
  1294→    if (raw === "true") return true;
  1295→    return p.default;
  1296→  }
  1297→  if (p.type === "path" || p.type === "text") {
  1298→    if (typeof raw === "string") return raw;
  1299→    return p.default;
  1300→  }
  1301→  if (p.type === "number") {
  1302→    const n = typeof raw === "number" ? raw : parseFloat(String(raw ?? ""));
  1303→    return Number.isFinite(n) ? n : p.default;
  1304→  }
  1305→  if (typeof raw === "string" && p.options?.includes(raw)) return raw;
  1306→  return p.default;
  1307→}

  1309→/** spec defaults overlaid with stored params (coerced). */
  1310→export function mergedParams(
  1311→  typeKey: string,
  1312→  stored: Record<string, ParamValue> | null | undefined
  1313→): Record<string, ParamValue> {
  1314→  const t = jobType(typeKey);
  1315→  const out: Record<string, ParamValue> = {};
  1316→  for (const p of t?.params ?? []) {
  1317→    out[p.key] = coerceParam(p, stored?.[p.key]);
  1318→  }
  1319→  return out;
  1320→}
